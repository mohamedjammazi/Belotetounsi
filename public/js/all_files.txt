function playTone(freq, duration, type = 'sine', gainVal = 0.15, delay = 0) {
  if (soundMuted) return;
  const ctx = getAudioContext();
  if (!ctx) return;
  const startAt = ctx.currentTime + delay;
  const osc = ctx.createOscillator();
  const gain = ctx.createGain();
  osc.type = type;
  osc.frequency.setValueAtTime(freq, startAt);
  gain.gain.setValueAtTime(gainVal, startAt);
  gain.gain.exponentialRampToValueAtTime(0.001, startAt + duration);
  osc.connect(gain);
  gain.connect(ctx.destination);
  osc.start(startAt);
  osc.stop(startAt + duration);
}

const IMA_INDEX_TABLE = [-1, -1, -1, -1, 2, 4, 6, 8, -1, -1, -1, -1, 2, 4, 6, 8];
const IMA_STEP_TABLE = [
7, 8, 9, 10, 11, 12, 13, 14, 16, 17, 19, 21, 23, 25, 28, 31, 34, 37, 41, 45,
50, 55, 60, 66, 73, 80, 88, 97, 107, 118, 130, 143, 157, 173, 190, 209, 230,
253, 279, 307, 337, 371, 408, 449, 494, 544, 598, 658, 724, 796, 876, 963,
1060, 1166, 1282, 1411, 1552, 1707, 1878, 2066, 2272, 2499, 2749, 3024, 3327,
3660, 4026, 4428, 4871, 5358, 5894, 6484, 7132, 7845, 8630, 9493, 10442,
11487, 12635, 13899, 15289, 16818, 18500, 20350, 22385, 24623, 27086, 29794, 32767
];
function encodeAdpcm4(floatSamples) {
  let pred = 0, index = 0;
  const out = new Uint8Array(Math.ceil(floatSamples.length / 2));
  let outIdx = 0, highNibble = false, byteVal = 0;
  for (let i = 0; i < floatSamples.length; i++) {
    const s = Math.max(-32768, Math.min(32767, Math.round(floatSamples[i] * 32767)));
    let diff = s - pred;
    let step = IMA_STEP_TABLE[index];
    let nibble = 0;
    if (diff < 0) { nibble = 8; diff = -diff; }
    let vpdiff = step >> 3;
    if (diff >= step) { nibble |= 4; diff -= step; vpdiff += step; }
    step >>= 1;
    if (diff >= step) { nibble |= 2; diff -= step; vpdiff += step; }
    step >>= 1;
    if (diff >= step) { nibble |= 1; vpdiff += step; }
    if (nibble & 8) pred -= vpdiff; else pred += vpdiff;
    pred = Math.max(-32768, Math.min(32767, pred));
    index = Math.max(0, Math.min(88, index + IMA_INDEX_TABLE[nibble]));
    if (!highNibble) { byteVal = nibble & 0x0F; highNibble = true; }
    else { byteVal |= (nibble & 0x0F) << 4; out[outIdx++] = byteVal; highNibble = false; }
  }
  if (highNibble) out[outIdx++] = byteVal;
  let binary = '';
  for (let i = 0; i < out.length; i++) binary += String.fromCharCode(out[i]);
  return btoa(binary);
}
function decodeAdpcm4(base64Str) {
  const binary = atob(base64Str);
  const len = binary.length;
  const out = new Float32Array(len * 2);
  let pred = 0, index = 0, outIdx = 0;
  for (let i = 0; i < len; i++) {
    const b = binary.charCodeAt(i);
    for (let shift = 0; shift <= 4; shift += 4) {
      const nibble = (b >> shift) & 0x0F;
      let step = IMA_STEP_TABLE[index];
      let vpdiff = step >> 3;
      if (nibble & 4) vpdiff += step;
      if (nibble & 2) vpdiff += step >> 1;
      if (nibble & 1) vpdiff += step >> 2;
      if (nibble & 8) pred -= vpdiff; else pred += vpdiff;
      pred = Math.max(-32768, Math.min(32767, pred));
      index = Math.max(0, Math.min(88, index + IMA_INDEX_TABLE[nibble]));
      out[outIdx++] = pred / 32768;
    }
  }
  return out;
}

  }
}

function playIncomingVoicePacket(pkt) {
  if (soundMuted || !pkt || !pkt.data) return;
  const ctx = getAudioContext();
  if (!ctx) return;

    const samples = decodeAdpcm4(pkt.data);
    const sr = Number(pkt.sampleRate) || 6000;
    const audioBuf = ctx.createBuffer(1, samples.length, sr);
    audioBuf.getChannelData(0).set(samples);

  function bindCarousel() {
    var vp = document.getElementById('carViewport');
    if (!vp || vp.dataset.bound) return;
    vp.dataset.bound = '1';
    var startX = 0, dx = 0, dragging = false;
    function down(x) { dragging = true; startX = x; dx = 0; }
    function move(x) { if (dragging) dx = x - startX; }
    function up() { if (!dragging) return; dragging = false; if (dx > 45) goSlide(state.carIdx - 1, true); else if (dx < -45) goSlide(state.carIdx + 1, true); }
    vp.addEventListener('touchstart', function (e) { down(e.touches[0].clientX); }, { passive: true });
    vp.addEventListener('touchmove', function (e) { move(e.touches[0].clientX); }, { passive: true });
    vp.addEventListener('touchend', up);
    vp.addEventListener('mousedown', function (e) { e.preventDefault(); down(e.clientX); });
    window.addEventListener('mousemove', function (e) { move(e.clientX); });
    window.addEventListener('mouseup', up);
    var pv = document.getElementById('carPrev'), nx = document.getElementById('carNext');
    if (pv) pv.onclick = function () { goSlide(state.carIdx - 1, true); };
    if (nx) nx.onclick = function () { goSlide(state.carIdx + 1, true); };
  }

  const deck = [];

    for (const rank of ['7', '8', '9', '10', 'J', 'Q', 'K', 'A']) deck.push({ suit, rank, id: `${rank}_${suit}` });
  }
  for (let i = deck.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [deck[i], deck[j]] = [deck[j], deck[i]]; }
  return deck;
}

function processLocalBid(r, seat, action, suit, value) {
  const mySeat = r.mySeat >= 0 ? r.mySeat : 0;

function getCardInnerElements(card) {
  const dr = beloteRankLabel(card.rank);

  const corners = `

  `;
  let body;
  if (card.rank === 'K' || card.rank === 'Q' || card.rank === 'J' || card.rank === 'R' || card.rank === 'D' || card.rank === 'V') {
    body = `<div class="court">${getCourtArt(card.rank, card.suit)}</div>`;

    body = `<div class="pip-ace">${getSuitSvg(card.suit, 'ace-suit-svg')}</div>`;

function computeLocalTrickWinner(trick, trumpSuit) {
  if (!trick || !trick.length) return null;
  const ledSuit = trick[0].card.suit; let best = trick[0];
  for (let i = 1; i < trick.length; i++) {
    const c = trick[i];
    const bT = best.card.suit === trumpSuit, cT = c.card.suit === trumpSuit;
    if (cT && !bT) best = c;
    else if (cT && bT) { if (POWER_TRUMP[c.card.rank] > POWER_TRUMP[best.card.rank]) best = c; }
    else if (!cT && !bT && c.card.suit === ledSuit) { if (POWER_NORMAL[c.card.rank] > POWER_NORMAL[best.card.rank]) best = c; }
  }
  return best;
}
function trickPointsOf(trick, trumpSuit) {
  return (trick || []).reduce((s, p) => s + (p.card.suit === trumpSuit ? POINTS_TRUMP[p.card.rank] : POINTS_NORMAL[p.card.rank]), 0);
}

  return {
    dx: Math.round((sr.left + sr.width / 2) - (centerRect.left + centerRect.width / 2)),
    dy: Math.round((sr.top + sr.height / 2) - (centerRect.top + centerRect.height / 2))
  };
}

/* ==========================================================================
   حساب حجم الكف — ثابت رياضياً على مقاس 8 أوراق دائماً
   حتى لا يتضخم الكف أو يقفز ارتفاعه عند رمي الأوراق أثناء اللعب
========================================================================== */
const HAND_STEP = 0.68; // 1 - 0.32
let _lastHandCW = 0, _lastAvailW = -1;
function layoutHandSize(hand, n) {
  const wrap = hand.parentElement;
  const availRaw = wrap ? wrap.getBoundingClientRect().width : hand.getBoundingClientRect().width;
  const avail = Math.max(160, Math.floor(availRaw) - 10);
  const rootCW = resolvedCardWidth();
  const refCount = Math.max(8, n || 8); // دائماً نحسب على 8 أوراق ليبقى الحجم والارتفاع ثابتين طوال الطرح
  const denom = 1 + HAND_STEP * (refCount - 1);
  let cw = Math.min(rootCW, (avail * 0.985) / denom);
  cw = Math.max(44, Math.floor(cw));
  if (cw !== _lastHandCW || avail !== _lastAvailW) {
    _lastHandCW = cw;
    _lastAvailW = avail;
    hand.style.setProperty('--card-w', cw + 'px');
  }
  return cw;
}

  const availH = Math.max(150, core.clientHeight - 6);
  const availW = Math.min(window.innerWidth * 0.92, core.clientWidth > 200 ? core.clientWidth - 6 : 500);
  const needH = box.scrollHeight;
  const needW = box.scrollWidth;
  if (needH > availH || needW > availW) {
    const s = Math.max(0.58, Math.min(availH / Math.max(1, needH), availW / Math.max(1, needW), 1));
    box.style.transform = `translate(-50%,-50%) scale(${s.toFixed(3)})`;
  }
}

function enhanceTurnUI(room, mySeat) {

  if (typeof room.turnRemainingMs === 'number') _turnDeadline = Date.now() + room.turnRemainingMs;

function paintTurnBanner() {
  const room = _turnRoom, el = document.getElementById('turnBanner');
  if (!room || !el) return;
  const mySeat = room.mySeat >= 0 ? room.mySeat : 0;
  const live = room.status === 'playing' || room.status === 'bidding';
  if (!live) { el.className = ''; el.textContent = ''; el.style.removeProperty('--tp'); return; }
  const mine = room.turnSeat === mySeat;
  const remMs = Math.max(0, _turnDeadline - Date.now());
  const remSec = Math.ceil(remMs / 1000);
  const late = remSec <= 15;
  const who = room.seats[room.turnSeat] ? room.seats[room.turnSeat].name : '...';
  const phase = room.status === 'bidding' ? 'المزايدة' : 'اللعب';
  const first = room.seats[room.firstPlayerSeat] ? ` • 🥇 ${room.seats[room.firstPlayerSeat].name}` : '';
  el.className = 'show' + (mine ? ' mine' : '') + (late ? ' late' : '');
  el.style.setProperty('--tp', Math.max(0, Math.min(100, (remMs / 60000) * 100)) + '%');
  el.innerHTML = `<span class="tb-main">${mine ? '⚡ دورك الآن!' : `⏳ دور <b>${escapeHtml(who)}</b>`}</span><small class="tb-sub">${phase} • ⏰ ${remSec}ث${escapeHtml(first)}</small>`;
  const slot = document.getElementById('slot' + getRelativePosition(room.turnSeat, mySeat));
  document.querySelectorAll('.player-slot.late').forEach(x => x.classList.remove('late'));
  if (slot && late) slot.classList.add('late');
  if (room.status === 'bidding') {
    const actingTeam = (mySeat % 2 === 0) ? 1 : 2;
    const open = isReactionWindowOpen(room, actingTeam);
    const rs = document.getElementById('reactSecs');
    if (rs && room.reactionUntil) rs.textContent = `(${Math.max(0, Math.ceil((room.reactionUntil - Date.now()) / 1000))}ث)`;
    if (!open && !mine) {
      const row = document.getElementById('snatchRow');
      if (row && row.style.display !== 'none') { row.style.display = 'none'; row.innerHTML = ''; }
    }
  }
}
